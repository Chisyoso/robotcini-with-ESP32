#include <ESP8266WiFi.h>
#include <ESP8266WebServer.h>

#include <Wire.h>
#include <Adafruit_GFX.h>
#include <Adafruit_SSD1306.h>

// =======================
// WIFI
// =======================

const char* ssid = "TU_WIFI";
const char* password = "TU_PASSWORD";

// =======================
// OLED
// =======================

#define SCREEN_WIDTH 128
#define SCREEN_HEIGHT 64
#define OLED_RESET -1

Adafruit_SSD1306 l(
  SCREEN_WIDTH,
  SCREEN_HEIGHT,
  &Wire,
  OLED_RESET
);

// =======================
// SERVER
// =======================

ESP8266WebServer server(80);

// =======================
// VARIABLES
// =======================

int emotion = 0;

// 0 normal
// 1 feliz
// 2 cute
// 3 triste
// 4 angry
// 5 dormido
// 6 hablando
// 7 pensando

bool talking = false;
bool thinking = false;

unsigned long lastAction = 0;
unsigned long blinkTimer = 0;
bool blinkNow = false;

// =======================
// OJOS
// =======================

void openEyes(){

  l.fillCircle(58,22,3,WHITE);
  l.fillCircle(70,22,3,WHITE);

}

void closedEyes(){

  l.drawLine(54,22,62,22,WHITE);
  l.drawLine(66,22,74,22,WHITE);

}

void angryEyes(){

  l.drawLine(54,18,62,24,WHITE);
  l.drawLine(66,24,74,18,WHITE);

}

void sleepEyes(){

  l.drawLine(54,24,62,24,WHITE);
  l.drawLine(66,24,74,24,WHITE);

  l.drawLine(60,16,64,10,WHITE);
  l.drawLine(66,16,70,10,WHITE);

}

// =======================
// BOCAS
// =======================

void mouthNormal(){

  l.drawLine(56,44,64,48,WHITE);
  l.drawLine(64,48,72,44,WHITE);

}

void mouthHappy(){

  l.drawCircle(64,42,10,WHITE);

  l.fillRect(52,28,24,12,BLACK);

}

void mouthSad(){

  l.drawLine(56,50,64,44,WHITE);
  l.drawLine(64,44,72,50,WHITE);

}

void mouthCute(){

  l.drawLine(58,42,64,48,WHITE);
  l.drawLine(64,48,70,42,WHITE);

  l.drawLine(64,48,64,52,WHITE);

}

void mouthTalk(){

  l.fillRoundRect(56,38,16,14,4,WHITE);

}

// =======================
// DIBUJAR
// =======================

void drawFace(){

  l.clearDisplay();

  bool sleepMode =
  (millis() - lastAction > 10000);

  if(sleepMode){

    emotion = 5;

  }

  // OJOS

  if(emotion == 4){

    angryEyes();

  }
  else if(emotion == 5){

    sleepEyes();

  }
  else{

    if(blinkNow){

      closedEyes();

    }else{

      openEyes();

    }

  }

  // BOCA

  if(talking){

    mouthTalk();

  }
  else{

    switch(emotion){

      case 0:
        mouthNormal();
      break;

      case 1:
        mouthHappy();
      break;

      case 2:
        mouthCute();
      break;

      case 3:
        mouthSad();
      break;

      case 4:
        mouthSad();
      break;

      case 5:
        mouthNormal();
      break;

      case 7:
        mouthCute();
      break;

      default:
        mouthNormal();
      break;

    }

  }

  // PENSANDO

  if(thinking){

    l.fillCircle(94,20,2,WHITE);
    l.fillCircle(100,16,3,WHITE);
    l.fillCircle(108,12,4,WHITE);

  }

  l.display();

}

// =======================
// ROUTES
// =======================

void handleEmotion(){

  String e = server.arg("e");

  lastAction = millis();

  if(e == "happy"){
    emotion = 1;
  }
  else if(e == "cute"){
    emotion = 2;
  }
  else if(e == "sad"){
    emotion = 3;
  }
  else if(e == "angry"){
    emotion = 4;
  }
  else{
    emotion = 0;
  }

  drawFace();

  server.send(200,"text/plain","ok");

}

void handleTalk(){

  talking = true;

  lastAction = millis();

  drawFace();

  server.send(200,"text/plain","talking");

}

void handleStopTalk(){

  talking = false;

  drawFace();

  server.send(200,"text/plain","stop");

}

void handleThinking(){

  thinking = true;

  lastAction = millis();

  drawFace();

  server.send(200,"text/plain","thinking");

}

void handleStopThinking(){

  thinking = false;

  drawFace();

  server.send(200,"text/plain","stop");

}

// =======================
// WIFI
// =======================

void connectWiFi(){

  WiFi.mode(WIFI_STA);

  WiFi.begin(ssid,password);

  Serial.println("");
  Serial.println("Conectando WiFi");

  while(WiFi.status() != WL_CONNECTED){

    delay(500);

    Serial.print(".");

  }

  Serial.println("");
  Serial.println("WiFi conectado");

  Serial.print("IP: ");
  Serial.println(WiFi.localIP());

}

// =======================
// SETUP
// =======================

void setup(){

  Serial.begin(115200);

  l.begin(
    SSD1306_SWITCHCAPVCC,
    0x3C
  );

  l.clearDisplay();
  l.display();

  connectWiFi();

  server.on("/emotion",handleEmotion);

  server.on("/talk",handleTalk);

  server.on("/stoptalk",handleStopTalk);

  server.on("/thinking",handleThinking);

  server.on("/stopthinking",handleStopThinking);

  server.begin();

  lastAction = millis();

}

// =======================
// LOOP
// =======================

void loop(){

  server.handleClient();

  // PARPADEO

  if(millis() - blinkTimer > 4000){

    blinkNow = true;

    drawFace();

    delay(120);

    blinkNow = false;

    blinkTimer = millis();

  }

  drawFace();

}
