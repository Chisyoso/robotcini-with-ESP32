// =======================

// =======================

// =======================
// VARIABLES
// =======================

unsigned long t;

}

void bw(){ // cute
  l.drawLine(54, 40, 59, 50, WHITE);
  l.drawLine(74, 40, 69, 50, WHITE);

  l.drawLine(69, 50, 64, 45, WHITE);
  l.drawLine(59, 50, 64, 45, WHITE);

  l.drawLine(54, 39, 59, 49, WHITE);
  l.drawLine(74, 39, 69, 49, WHITE);
}

void sad(){ // triste
  l.drawLine(56, 48, 64, 42, WHITE);
  l.drawLine(64, 42, 72, 48, WHITE);
}

void angryEyes(){
  l.drawLine(55, 20, 61, 24, WHITE);
  l.drawLine(67, 24, 73, 20, WHITE);
}

// =======================
// DIBUJAR
// =======================

void drawFace(){

  l.clearDisplay();

  switch(e){

    case 0:
      oa();
      bf();
    break;

    case 1:
      oa();
      bmf();
    break;

    case 2:
      oc();
      bw();
    break;

    case 3:
      oa();
      sad();
    break;

    case 4:
      angryEyes();
      sad();
    break;

  }

  l.display();
}

// =======================
// RUTAS
// =======================

void handleRoot(){

  server.send(200, "text/html", html);

}

void handleEmotion(){

  String emotion = server.arg("e");

  if(emotion == "happy"){
    e = 1;
  }
  else if(emotion == "cute"){
    e = 2;
  }
  else if(emotion == "sad"){
    e = 3;
  }
  else if(emotion == "angry"){
    e = 4;
  }
  else{
    e = 0;
  }

  te = millis();

  drawFace();

  server.send(200, "text/plain", "ok");
}

// =======================
// WIFI CONNECT
// =======================

void connectWiFi(){

  WiFi.mode(WIFI_STA);

  WiFi.begin(ssid, password);

  Serial.println("");
  Serial.println("Conectando a WiFi");

  while(WiFi.status() != WL_CONNECTED){

    delay(500);

    Serial.print(".");

  }

  Serial.println("");
  Serial.println("WiFi conectado");
  Serial.println("");

  Serial.print("IP: ");
  Serial.println(WiFi.localIP());

}

// =======================
// SETUP
// =======================

void setup(){

  Serial.begin(115200);

  // OLED
  l.begin(SSD1306_SWITCHCAPVCC, 0x3C);

  l.clearDisplay();
  l.setTextColor(WHITE);
  l.setTextSize(1);

  drawFace();

  // WIFI
  connectWiFi();

  // SERVER
  server.on("/", handleRoot);

  server.on("/emotion", handleEmotion);

  server.begin();

  Serial.println("Servidor iniciado");

}

// =======================
// LOOP
// =======================

void loop(){

  server.handleClient();

  // volver a normal luego de tiempo
  if(millis() - te > 10000){

    e = 0;

  }

  // parpadeo
  if(millis() - t > 4000){

    l.clearDisplay();

    oc();

    switch(e){

      case 0:
        bf();
      break;

      case 1:
        bmf();
      break;

      case 2:
        bw();
      break;

      case 3:
        sad();
      break;

      case 4:
        sad();
      break;

    }

    l.display();

    delay(120);

    t = millis();

  }

  drawFace();

}
